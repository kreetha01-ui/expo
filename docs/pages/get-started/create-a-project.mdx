System requirements:

- [Node.js (LTS)](https://nodejs.org/en/).
- macOS, Windows (Powershell and [WSL 2](https://expo.fyi/wsl)), and Linux are supported.

We recommend starting with the default project created by `create-expo-app`. The default project includes example code to help you get started.

To create a new project, run the following command:

<Terminal cmd={['$ npx create-expo-app@latest']} />

> You can choose a different template by adding the [`--template` option](/more/create-expo/#--template).

## Next step

You have a project. Now it's time to set up your development environment so that you can start developing.

        // LINE Notify functions
        function saveToken() {
            const token = document.getElementById('lineToken').value;
            if (token) {
                localStorage.setItem('lineNotifyToken', token);
                return token;
            }
            return null;
        }

        function getToken() {
            return localStorage.getItem('lineNotifyToken');
        }

        async function sendLineNotify(message, token) {
            try {
                const response = await fetch('https://notify-api.line.me/api/notify', {
                    method: 'POST',
                    headers: {
                        'Authorization': `Bearer ${token}`,
                        'Content-Type': 'application/x-www-form-urlencoded',
                    },
                    body: `message=${encodeURIComponent(message)}`
                });

                if (response.ok) {
                    return { success: true };
                } else {
                    const error = await response.text();
                    return { success: false, error: error };
                }
            } catch (error) {
                return { success: false, error: error.message };
            }
        }

        async function sendReportToLine(data, reportId) {
            // Show loading state
            const submitBtn = document.querySelector('button[type="submit"]');
            const originalText = submitBtn.innerHTML;
            submitBtn.innerHTML = '<svg class="w-5 h-5 animate-spin mr-2" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M4 4v5h.582m15.356 2A8.001 8.001 0 004.582 9m0 0H9m11 11v-5h-.581m0 0a8.003 8.003 0 01-15.357-2m15.357 2H15"></path></svg>กำลังส่ง...';
            submitBtn.disabled = true;

            // Create LINE message format
            const lineMessage = `🚨 แจ้งไฟส่องสว่างดับ 🚨
                
📍 ตำแหน่ง: ${data.location}
⚡ ปัญหา: ${data.problemType}
📝 รายละเอียด: ${data.description || 'ไม่ระบุ'}
👤 ผู้แจ้ง: ${data.reporterName}
📱 เบอร์: ${data.phone}
🚨 ความเร่งด่วน: ${data.priority}
⏰ เวลา: ${data.timestamp}
🆔 รหัสรายงาน: ${reportId}

กรุณาดำเนินการตรวจสอบและแก้ไข`;

            const token = getToken();
            let success = false;

            if (token) {
                const result = await sendLineNotify(lineMessage, token);
                success = result.success;
                
                if (!success) {
                    console.error('LINE Notify Error:', result.error);
                }
            }

            // Show success message
            document.getElementById('reportId').textContent = reportId;
            document.getElementById('statusMessage').classList.remove('hidden');
            document.querySelector('.container > div:nth-child(2)').style.display = 'none';
            
            // Update success message based on LINE send result
            const statusTitle = document.querySelector('#statusMessage h3');
            const statusDesc = document.querySelector('#statusMessage p');
            
            if (success) {
                statusTitle.textContent = 'ส่งรายงานและแจ้ง LINE สำเร็จ! ✅';
                statusDesc.textContent = 'รายงานได้ถูกส่งไปยัง LINE และหน่วยงานที่เกี่ยวข้องแล้ว';
            } else if (token) {
                statusTitle.textContent = 'ส่งรายงานสำเร็จ แต่ LINE ส่งไม่ได้ ⚠️';
                statusDesc.textContent = 'รายงานบันทึกแล้ว แต่การส่ง LINE มีปัญหา กรุณาตรวจสอบ Token';
            } else {
                statusTitle.textContent = 'ส่งรายงานสำเร็จ! 📝';
                statusDesc.textContent = 'รายงานได้ถูกบันทึกแล้ว (ตั้งค่า LINE Token เพื่อส่งแจ้งเตือน)';
            }
            
            // Reset form
            document.getElementById('reportForm').reset();
            document.querySelectorAll('input[name="priority"]').forEach(r => {
                const label = r.closest('label');
                label.classList.remove('border-blue-500', 'bg-blue-50');
            });
            
            submitBtn.innerHTML = originalText;
            submitBtn.disabled = false;
            
            // Auto scroll to success message
            document.getElementById('statusMessage').scrollIntoView({ behavior: 'smooth' });
        }

        // Load saved token on page load

        // Reset form function
        function resetForm() {
            document.getElementById('statusMessage').classList.add('hidden');

        }

            if (e.target.closest('.text-center')) {
                const resetBtn = document.createElement('button');
                resetBtn.className = 'mt-4 bg-blue-500 text-white px-6 py-2 rounded-lg hover:bg-blue-600 transition-colors';
                resetBtn.textContent = 'แจ้งเหตุใหม่';
                resetBtn.onclick = resetForm;
                
                if (!this.querySelector('button')) {
                    this.querySelector('.text-center').appendChild(resetBtn);
                }
            }
